// ----------------------------------------------------
// 4-FSR Posture Tracker (Seat + 3 Back Sensors)
// Threshold-filtered version for stable readings
// ----------------------------------------------------

const int fsrPins[4] = {A0, A1, A2, A3};
int fsrValues[4];

void setup() {
  Serial.begin(9600);
  for (int i = 0; i < 4; i++) pinMode(fsrPins[i], INPUT);
  delay(1000); // Give Serial time to start
}

void loop() {
  // Read all sensors
  for (int i = 0; i < 4; i++) {
    fsrValues[i] = analogRead(fsrPins[i]);

    // Apply threshold filtering
    if (i == 0) {
      // Seat sensor (A0)
      if (fsrValues[i] < 500) fsrValues[i] = 0;
    } else {
      // Back sensors (A1–A3)
      if (fsrValues[i] < 250) fsrValues[i] = 0;
    }
  }

  // Send all 4 as comma-separated numbers
  Serial.print(fsrValues[0]);
  Serial.print(",");
  Serial.print(fsrValues[1]);
  Serial.print(",");
  Serial.print(fsrValues[2]);
  Serial.print(",");
  Serial.println(fsrValues[3]);

  delay(200);
}
